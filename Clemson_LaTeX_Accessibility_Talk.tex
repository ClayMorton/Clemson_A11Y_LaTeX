  % with no warning (ISO 14289-2:2024 8.11). The block goes above

  % a package report at the end of the log. The package comes last because it

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{itemize}
        \item \texttt{lang}: the whole PDF
        \item \texttt{pdfstandard}: PDF/UA-2
        \item \texttt{tagging}: on, or the build stops
        \item \texttt{check-tagging-status}: a report in the log
        \item The package comes last
      \end{itemize}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small

\begin{frame}
  \frametitle{What the package does}
  % Everything else is plain LaTeX.
  \begin{itemize}
    \item Stops the build when tagging is off
    \item Puts every formula's MathML in the tag tree
    \item Title H1, every heading one level down
    \item Figures and tables where they are written, caption first
    \item Scope and headers on every table cell
    \item Web and email links underlined; \texttt{\textbackslash email}
    \item Notes, other languages, theorems and proofs tagged
    \item Text ragged right (\texttt{[justified]} to keep it justified)
  \end{itemize}

  \medskip

  Everything else is plain LaTeX.
\end{frame}

\begin{frame*}
  \frametitle{Title, authors, headings}
  % A screen reader moves by headings, so the title is the only H1 and no
  % level is skipped (Clemson Digital Accessibility, Headings; PDF/UA-1
  % 7.4.2). In a document: title H1, \section H2, \subsection H3. In this
  % deck: title H1, frame title H2. Only the title text is the heading;
  % authors and date are paragraphs. \thanks adds a footnote to a name and
  % stays out of the metadata. Nothing to write: the package sets the levels.
  One H1, no skipped level.

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{itemize}
        \item Document: title H1, \verb|\section| H2, \verb|\subsection| H3
        \item Deck: title H1, frame title H2
        \item \verb|\thanks|: a footnote, kept out of the metadata
      \end{itemize}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\title{Accessible LaTeX, quickly}
\author{First Author\thanks{Corresponding
  author.} \and Second Author}
\date{\today}
\begin{document}
\maketitle
\section{How to use this file}
\subsection{Links}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{What a formula means}
  % MathML records how a formula looks, not what it means: an x with a bar is
  % "x bar" whether it is a mean or a complex conjugate. \MathMLintent adds
  % the meaning, \MathMLarg names the part it applies to (Math BPG 1.0,
  % intent). Keep the double braces: a scripted single atom loses the intent
  % with no warning.
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      ``Mean of x'', not ``x bar'':
      \begin{equation}
        \MathMLintent{mean($x)}{{\overline{\MathMLarg{x}{x}}}}
        = \frac{1}{n} \sum_{i=1}^{n} x_i
      \end{equation}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{equation}
  \MathMLintent{mean($x)}
    {{\overline{\MathMLarg{x}{x}}}}
  = \frac{1}{n} \sum_{i=1}^{n} x_i
\end{equation}
\end{verbatim}
    \end{column}
  \end{columns}

  \medskip

\begin{frame*}
  \frametitle{A header row}
  % LaTeX marks header cells, but Acrobat and PAC do not read its scope
  % classes and nothing ties a cell to its headers; the package writes Scope
  % and Headers on every cell (ISO 32000-2:2020 14.8.4.8.3, 14.8.5.7). Declare
  % the header cells before \begin{tabular}: each cell in row 1 becomes a TH
  % with column scope.
  Declare the header cells; the package writes Scope and Headers.

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{table}
        \caption{Header row}\label{tab:headerrow}
        \tagpdfsetup{table/header-rows={1}}
        \begin{tabular}{lrr}
          \toprule Item & Count & Share \\ \midrule
          Alpha & 12 & 40\% \\
          Beta  & 18 & 60\% \\ \bottomrule
        \end{tabular}
      \end{table}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{table}
  \caption{Header row}
  \tagpdfsetup{table/header-rows={1}}
  \begin{tabular}{lrr}
    \toprule Item & Count & Share \\
    \midrule
    Alpha & 12 & 40\% \\
    Beta  & 18 & 60\% \\ \bottomrule
  \end{tabular}
\end{table}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{A complex table: two header rows, merged cells}
  % Both header rows are declared. \multicolumn cells get their column span
  % by themselves; table/multirow=2 makes the corner cell span both rows, so
  % the blank cell under it is not an empty header.
  \begin{columns}
    \begin{column}[t]{.44\textwidth}
      \begin{table}
        \caption{Two-level header}\label{tab:complex}
        \tagpdfsetup{table/header-rows={1,2}, table/header-columns={1}}
        \begin{tabular}{lcccc}
          \toprule
          \tagpdfsetup{table/multirow=2}%
          Method & \multicolumn{2}{c}{Training} & \multicolumn{2}{c}{Test} \\
                 & Prec. & Rec. & Prec. & Rec. \\
          \midrule
          Baseline & 0.74 & 0.68 & 0.71 & 0.64 \\
          Proposed & 0.91 & 0.86 & 0.88 & 0.81 \\ \bottomrule
        \end{tabular}
      \end{table}
    \end{column}

\tagpdfsetup{table/header-rows={1,2},
  table/header-columns={1}}
