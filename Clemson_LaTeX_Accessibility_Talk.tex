\begin{frame*}
  \frametitle{The preamble}
  An untagged PDF looks normal and fails the check, and LaTeX
  writes one with no warning (ISO 14289-2:2024 8.11; LaTeX tagging status list).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{itemize}
        \item \texttt{lang}: the language of the whole PDF
        \item \texttt{pdfstandard}: declares PDF/UA-2
        \item \texttt{tagging}: turns tagging on; the package stops the build
          when it is off
        \item \texttt{check-tagging-status}: a package report at the end of
          the log
        \item The package comes after the other packages: it loads hyperref,
          which must come last
      \end{itemize}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\DocumentMetadata{
  lang        = en-US,
  pdfstandard = ua-2,
  tagging     = on,
  check-tagging-status,
}
\documentclass{article}
\usepackage{booktabs}
\usepackage{clemson}
\graphicspath{{resources/}}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame}
  \frametitle{What the package does}
  \begin{itemize}
    \item Stops the build when tagging is off; warns on \texttt{tagging=draft}
    \item Loads hyperref, unicode-math and babel; writes every formula's
      MathML into the tag tree
    \item Tags the title as the only H1 and moves every heading one level down
    \item Keeps every figure and table where it is written, caption first
    \item Writes the scope and the headers of every table cell
    \item Underlines web and email links; defines \texttt{\textbackslash email}
    \item Tags footnotes as notes and foreign phrases with their language;
      defines theorem, lemma, proof and friends
    \item Sets the text ragged right; \texttt{[justified]} keeps it justified
  \end{itemize}

  \medskip

  Everything else is plain LaTeX: nothing to learn, nothing to work around.
\end{frame}

\begin{frame*}
  \frametitle{Title, authors, headings}
  A screen reader uses headings to move around, so the title is the only H1
  and no level is skipped (Clemson Digital Accessibility, Headings; PDF/UA-1
  7.4.2).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      In a document: title H1, \verb|\section| H2, \verb|\subsection| H3. In
      this deck: title H1, frame title H2. Only the title text is
      the heading; authors and date are paragraphs.

      \medskip

      \verb|\thanks| adds a footnote to a name; LaTeX leaves it out of the
      metadata. The package sets the levels; nothing to write.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\title{Accessible LaTeX, quickly}
\author{First Author\thanks{Corresponding
  author.} \and Second Author}
\date{\today}
\begin{document}
\maketitle
\section{How to use this file}
\subsection{Links}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Emphasis and notes}
  Emphasis, notes, links and lists have their own tags, so a
  screen reader says what each one is, not just its letters (ISO 32000-2:2020
  14.8.4.7, 14.8.4.8.2).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      Use \verb|\emph| for \emph{emphasis} and \verb|\strong| for
      \strong{important text}; both are tagged, while \verb|\textbf| only
      changes how letters are drawn.\footnote{A footnote is tagged as a note;
      its mark links to it and the note links back.} A long aside goes in an
      endnote, printed near the end of a document.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
Use \emph{emphasis} and
\strong{important text}.
A footnote.\footnote{A note at
  the foot of the page.}
An aside.\endnote{Printed near
  the end.}
...
\printendnotes
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Links}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      A link has its own text, so name the destination:
      \href{https://www.clemson.edu/accessibility/}{Clemson accessibility}, or
      print the address, \url{https://verapdf.org/}. Write an email address
      with \verb|\email|: \email{first.last@clemson.edu}.

      \medskip

      The package underlines web and email links in the text itself, so every
      viewer shows it. Links inside the document, such as citations and
      cross-references, are not underlined (Clemson Digital Accessibility,
      Links).
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\href{https://www.clemson.edu/accessibility/}
  {Clemson accessibility}
\url{https://verapdf.org/}
\email{first.last@clemson.edu}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Cross-references}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      Write \verb|\autoref| rather than ``the section above'': the whole
      phrase is one link, and a screen reader reads ``Theorem 1'', not a bare
      number. See \autoref{thm:one} on page~\pageref{thm:one},
      \autoref{alg:max} and Equation~\eqref{eq:gauss}.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
See \autoref{thm:one} on
page~\pageref{thm:one},
\autoref{alg:max} and
Equation~\eqref{eq:gauss}.
\end{verbatim}
      Theorem-like blocks work too: the package names each one for
      \verb|\autoref|.
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Quotations, special characters, code}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}

      BlockQuote:

      \end{quote}

    \end{column}

  part of a document is.
\end{quote}

\end{verbatim}

\begin{frame*}
  \frametitle{Inline and displayed formulas}
  Every formula goes into the PDF as MathML, so a screen reader reads
  its structure, not a string of symbols (ISO 14289-2:2024 8.2.5.29.1; PDF
  Association Math BPG 1.0).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      Inline mathematics is part of the sentence: $e^{i\pi} + 1 = 0$. A displayed
      equation takes a label so that it can be cited:
      \begin{equation}\label{eq:gauss}
        \int_{-\infty}^{\infty} e^{-x^2}\,dx = \sqrt{\pi}
      \end{equation}
      Equation~\eqref{eq:gauss} is cited by number, and the number is a link.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
Inline mathematics is part of the
sentence: $e^{i\pi} + 1 = 0$.
\begin{equation}\label{eq:gauss}
  \int_{-\infty}^{\infty}
    e^{-x^2}\,dx = \sqrt{\pi}
\end{equation}
Equation~\eqref{eq:gauss} is
cited by number.
\end{verbatim}
      LaTeX writes the MathML and the package puts it in the tag tree.
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Aligned steps and cases}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      Matrices, cases, sums and aligned steps come from amsmath, which the
      package loads:
      \begin{align}
        (a+b)^2 &= a^2 + 2ab + b^2 \\
                &= a^2 + b^2 + 2ab
      \end{align}
      \begin{equation}
        f(x) = \begin{cases} x^2 & x \ge 0 \\ -x & x < 0 \end{cases}
      \end{equation}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{align}
  (a+b)^2 &= a^2 + 2ab + b^2 \\
          &= a^2 + b^2 + 2ab
\end{align}
\begin{equation}
  f(x) = \begin{cases}
    x^2 & x \ge 0 \\ -x & x < 0
  \end{cases}
\end{equation}
\end{verbatim}
      Every line is a Formula with its own MathML. One limit: a numbered
      \texttt{multline} warns on every run, so use \texttt{multline*}.
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{What a formula means}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      MathML records how a formula looks, not what it means: an $x$ with a bar
      is ``x bar'' whether it is a mean or a complex conjugate.
      \verb|\MathMLintent| adds the meaning and \verb|\MathMLarg| names the
      part it applies to, so a reader says ``mean of x'' for
      \begin{equation}
        \MathMLintent{mean($x)}{{\overline{\MathMLarg{x}{x}}}}
        = \frac{1}{n} \sum_{i=1}^{n} x_i
      \end{equation}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{equation}
  \MathMLintent{mean($x)}
    {{\overline{\MathMLarg{x}{x}}}}
  = \frac{1}{n} \sum_{i=1}^{n} x_i
\end{equation}
\end{verbatim}
      Keep the double braces: a scripted single atom loses the intent with no
      warning (Math BPG 1.0, intent).
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Theorems and proofs}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{theorem}\label{thm:one}
        Every heading in a tagged PDF has a level.
      \end{theorem}
      \begin{proof}
        The package moves every heading one level down.
      \end{proof}
      The package defines theorem, lemma, proposition, corollary, definition,
      example, remark and algorithm; LaTeX tags each block with its head as a
      caption. The proof ends with the word QED, which a screen reader reads; the
      open square is not read.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{theorem}\label{thm:one}
  Every heading in a tagged PDF
  has a level.
\end{theorem}
\begin{proof}
  The package moves every heading
  one level down.
\end{proof}
\end{verbatim}
      No \verb|\newtheorem| line: the package defines them. No
      \verb|\qedsymbol|: the proof ends with QED as a word.
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Algorithms}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{algorithm}[Find the largest value]\label{alg:max}
        Input: $a_1,\dots,a_n$. Output: the largest of them.
        \begin{algorithmic}[1]
          \State $m \gets a_1$
          \For{$i \gets 2$ to $n$}
            \If{$a_i > m$} \State $m \gets a_i$ \EndIf
          \EndFor
          \State \Return $m$
        \end{algorithmic}
      \end{algorithm}
      The package's \texttt{algorithm} is a theorem-like block, not a float:
      the algorithm package is incompatible with tagging.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{algorithm}[Find the largest
  value]\label{alg:max}
  Input: $a_1,\dots,a_n$.
  \begin{algorithmic}[1]
    \State $m \gets a_1$
    \For{$i \gets 2$ to $n$}
      \If{$a_i > m$}
        \State $m \gets a_i$
      \EndIf
    \EndFor
    \State \Return $m$
  \end{algorithmic}
\end{algorithm}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

  A screen reader skips a picture unless alt text says what it
  shows (WCAG 2.1 1.1.1; ISO 32000-2:2020 14.8.4.8.5).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{figure}
        \caption{Cables in the store}\label{fig:cables}
        \includegraphics[width=.7\linewidth,alt={Coiled gray cables on a beige pegboard wall.}]{cables}
      \end{figure}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{figure}
  \caption{Cables in the store}
  \label{fig:cables}
  \includegraphics[width=.7\linewidth,
    alt={Coiled gray cables on a
      beige pegboard wall.}]{cables}
\end{figure}
\end{verbatim}

      printed below would contradict the tag order. Say what the picture shows
      and why it is here, never the file name.

  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{A header row}
  LaTeX marks header cells, but Acrobat and PAC do not read its
  scope classes and nothing ties a cell to its headers; the package writes
  Scope and Headers on every cell (ISO 32000-2:2020 14.8.4.8.3, 14.8.5.7).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{table}
        \caption{Header row}\label{tab:headerrow}
        \tagpdfsetup{table/header-rows={1}}
        \begin{tabular}{lrr}
          \toprule Item & Count & Share \\ \midrule
          Alpha & 12 & 40\% \\
          Beta  & 18 & 60\% \\ \bottomrule
        \end{tabular}
      \end{table}
      Declare the header cells before \verb|\begin{tabular}|: each cell in
      row 1 becomes a TH with column scope.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{table}
  \caption{Header row}
  \tagpdfsetup{table/header-rows={1}}
  \begin{tabular}{lrr}
    \toprule Item & Count & Share \\
    \midrule
    Alpha & 12 & 40\% \\
    Beta  & 18 & 60\% \\ \bottomrule
  \end{tabular}
\end{table}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{A header column}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{table}
        \caption{Header column}\label{tab:headercol}
        \tagpdfsetup{table/header-columns={1}}
        \begin{tabular}{lrrr}
          \toprule
          Mean   & 4.1 & 4.4 & 4.0 \\
          Median & 4.0 & 4.5 & 3.9 \\ \bottomrule
        \end{tabular}
      \end{table}
      Each cell in column 1 is a TH with row scope. Give both keys when a
      table has both; the corner cell then has both scopes.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\caption{Header column}
\tagpdfsetup{table/header-columns={1}}
\begin{tabular}{lrrr}
  \toprule
  Mean   & 4.1 & 4.4 & 4.0 \\
  Median & 4.0 & 4.5 & 3.9 \\ \bottomrule
\end{tabular}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{A complex table: two header rows, merged cells}
  \begin{columns}
    \begin{column}[t]{.44\textwidth}
      \begin{table}
        \caption{Two-level header}\label{tab:complex}
        \tagpdfsetup{table/header-rows={1,2}, table/header-columns={1}}
        \begin{tabular}{lcccc}
          \toprule
          \tagpdfsetup{table/multirow=2}%
          Method & \multicolumn{2}{c}{Training} & \multicolumn{2}{c}{Test} \\
                 & Prec. & Rec. & Prec. & Rec. \\
          \midrule
          Baseline & 0.74 & 0.68 & 0.71 & 0.64 \\
          Proposed & 0.91 & 0.86 & 0.88 & 0.81 \\ \bottomrule
        \end{tabular}
      \end{table}
      Both header rows are declared. \verb|\multicolumn| cells get their
      column span by themselves; \texttt{table/multirow=2} makes the corner
      cell span both rows, so the blank cell under it is not an empty header.
    \end{column}
    \begin{column}[t]{.53\textwidth}
\small
\begin{verbatim}
\tagpdfsetup{table/header-rows={1,2},
  table/header-columns={1}}
\begin{tabular}{lcccc}
  \toprule
  \tagpdfsetup{table/multirow=2}%

         & \multicolumn{2}{c}{Test} \\
         & Prec. & Rec. & Prec. & Rec. \\
  \midrule
  Baseline & 0.74 & 0.68 & 0.71 & 0.64 \\
  Proposed & 0.91 & 0.86 & 0.88 & 0.81 \\
  \bottomrule
\end{tabular}
\end{verbatim}
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Side by side is not a table}
  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{minipage}[t]{0.45\linewidth}\centering
        First Author\\ Clemson University
      \end{minipage}\hfill
      \begin{minipage}[t]{0.45\linewidth}\centering
        Second Author\\ Clemson University
      \end{minipage}

      \medskip

      A layout tabular is read row by row: ``First Author Second Author
      Clemson University Clemson University''. A minipage per author is read
      to its end, so each name comes with its own university.
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
\begin{minipage}[t]{0.45\linewidth}
  \centering
  First Author\\ Clemson University
\end{minipage}\hfill
\begin{minipage}[t]{0.45\linewidth}
  \centering
  Second Author\\ Clemson University
\end{minipage}
\end{verbatim}
      Also: caption above the table, notes as a paragraph after it, no empty
      cells, one table per group, never a table inside a table.
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Check the PDF}
  veraPDF tests the machine rules; five checks by hand cover
  what no script can judge (Clemson Digital Accessibility, PDF manual checks).

  \medskip

  \begin{columns}
    \begin{column}[t]{.42\textwidth}
      \begin{enumerate}
        \item Every alt text says what the picture shows and why it is here.
        \item Acrobat Pro, Check for accessibility: the title is the only H1,
          the sections start at H2, the reading order follows the page.
        \item Every table declared its header cells; the Table Editor shows
          them.
        \item Link text names the destination; Tab reaches every link.
        \item Nothing is said by color alone; 4.5:1 contrast for text, 3:1 for
          graphics.
      \end{enumerate}
    \end{column}
    \begin{column}[t]{.55\textwidth}
\small
\begin{verbatim}
latexmk -lualatex main.tex
verapdf --flavour ua2 --format text main.pdf
grep -n -A2 '^!\|tagpdf Warning\|clemson' \
  main.log
\end{verbatim}
      PASS or FAIL on the first line, then each failed clause. The log lines
      list errors, tagging warnings and every picture without alt text.
    \end{column}
  \end{columns}
\end{frame*}

\begin{frame*}
  \frametitle{Copy the kit and write}
  \begin{enumerate}
    \item Keep \verb|\DocumentMetadata| above \verb|\documentclass|; load
      your packages, then \verb|\usepackage{clemson}|.
    \item Write plain LaTeX: \verb|\section| in order, \verb|\autoref|,
      \verb|\emph| and \verb|\strong|.
    \item Give every picture \texttt{alt=} and every table its header cells;
      build with latexmk, then check with veraPDF and the five checks by hand.
  \end{enumerate}

  \medskip

  The kit: \url{https://github.com/ClayMorton/Clemson_A11Y_LaTeX}.
  \texttt{main.tex} is the blank starter, \texttt{example.tex} the worked
  example, \texttt{README.md} the instructions.
\end{frame*}
