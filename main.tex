% METADATA AND CLASS

% Keep this block above \documentclass: it turns tagging on and names the
% standard. Meets: ISO 14289-2:2024 8.11 (metadata), 8.2.5.29 (MathML).
\DocumentMetadata{
  % The language of the whole PDF, written to /Lang. Phrases in another
  % language are marked in the text; example.tex shows how.
  lang          = en-US,
  % Declares PDF/UA-2, the accessibility standard for PDF 2.0. veraPDF checks
  % the file against it with --flavour ua2.
  pdfstandard   = ua-2,
  % Turns tagging on. Without it LaTeX writes an untagged PDF and says
  % nothing; the clemson package stops the build instead.
  tagging       = on,
  % Writes the MathML of every formula as structure elements (Acrobat, NVDA)
  % and as an attached file (Foxit, Firefox). The LaTeX default is the file.
  tagging-setup = {math/setup={mathml-SE,mathml-AF}},
  % Appends a report on the loaded packages to the end of main.log; sections
  % 1 and 2 name packages to replace.
  check-tagging-status,
}
